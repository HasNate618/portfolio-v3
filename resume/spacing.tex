% bullet counts.

\setlength{\ResumeSubHeadingGap}{-7pt}
\setlength{\ResumeListEndGap}{-5pt}

\setlength{\ResumeItemGap}{-4pt}
\setlength{\ResumeSubHeadingGap}{-7pt}
\setlength{\ResumeListEndGap}{-5pt}
